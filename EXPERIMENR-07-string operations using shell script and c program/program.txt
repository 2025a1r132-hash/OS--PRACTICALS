# Experiment 7: String Operations Using Shell Script and C Program

# 1. Shell script to extract substring and find length

#!/bin/bash

echo "Enter a string:"
read str

echo "Enter starting position:"
read start

echo "Enter substring length:"
read len

echo "Substring:"
echo "$str" | cut -c${start}-$((start+len-1))

echo "Length of string: ${#str}"


// 2. C program to extract substring and find length

#include <stdio.h>
#include <string.h>

int main() {
    char str[100];
    int start, len, i;

    printf("Enter a string: ");
    scanf("%99s", str);

    printf("Enter starting position (0-based): ");
    scanf("%d", &start);

    printf("Enter substring length: ");
    scanf("%d", &len);

    printf("Substring: ");

    for (i = start; i < start + len && str[i] != '\0'; i++) {
        if (start >= 0 && len >= 0)
            printf("%c", str[i]);
    }

    printf("\nString length: %lu\n",
           (unsigned long)strlen(str));

    return 0;
}
